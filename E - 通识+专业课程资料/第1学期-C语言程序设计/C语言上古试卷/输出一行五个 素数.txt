 编写函数int isprime(int m)，判断某一个数是否是一个素数。调用该函数求解1~500之间的素数，并每行输出5个。
#include <stdio.h>
#include <math.h>
int isprime(int m)
{
	int i,flag=1;
	for(i=2;i<=sqrt(m);i++)
		if (m%i==0) 
		{
			flag=0;
			break;
		}
	return flag;
}
int main()
{
	int i,count=0;
	for(i=1;i<=500;i++)
		if(isprime(i)) 
		{
			printf("%5d",i);
			count++;
		 if(count%5==0) printf("\n");
		}
	return 0;
}
